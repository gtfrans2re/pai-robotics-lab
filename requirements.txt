# Python deps for the RL + edge-AI pillars (install inside the venv).

#
# Versions are left unpinned for portability; pin from your own `pip freeze`
# for an exactly reproducible env. Built/tested on Python 3.12, Ubuntu 24.04.

numpy
opencv-python

# RL
mujoco
pybullet
gymnasium

# Edge AI
ultralytics        # YOLOv11 (pulls torch; see README for the CPU-only torch swap)
openvino
